# 基于 LobeHub 的 AI 多模态交互平台 — 技术实施方案（细化版）

## Context

甲方要求基于 LobeHub 构建覆盖文本 / 图像 / 语音 / 视频的 AI 多模态交互平台，支持在线 API 与离线本地模型双模式，并具备企业级多用户管理和知识库能力。经全面代码分析，13 项需求中 9 项可直接复用现有代码，需重点开发的是 RBAC 激活、用户层级管理、视频到视频 3 项。推荐分三期交付：一期复用现有能力快速出可见功能，二期新增企业管理域，三期 PoC 高差异能力。

## 二、概念混淆预警

1. `sessionGroups` / `chatGroups` ≠ 企业用户分组 — 前者是会话分组 / 多 Agent 协作，后者是组织管理，必须单独建模
2. 视频页面存在 ≠ v2v 已完成 — 现有只支持 t2v 和 i2v（首帧），无视频输入链路
3. ComfyUI provider ≠ 离线视频闭环 — 视频工作流生态不成熟，需 PoC
4. RBAC 表存在 ≠ 权限生效 — 4 张表已创建但无中间件调用

### 4.1 RBAC 权限中间件激活

#### 4.1.1 新建 RBAC 中间件

**文件**：`src/libs/trpc/middleware/rbacAuth.ts`（新建）

参考 [userAuth.ts](src/libs/trpc/middleware/userAuth.ts)（17 行）的精确模式：

```typescript
import { TRPCError } from '@trpc/server';
import { getServerDB } from '@/database/core/db-adaptor';
import { RbacModel } from '@lobechat/database/models/rbac';
import { trpc } from '../lambda/init';

export const requirePermission = (permissionCode: string) =>
  trpc.middleware(async (opts) => {
    const { ctx } = opts;
    if (!ctx.userId) throw new TRPCError({ code: 'UNAUTHORIZED' });

    const serverDB = await getServerDB();
    const rbacModel = new RbacModel(serverDB, ctx.userId);
    const hasPermission = await rbacModel.hasPermission(permissionCode);

    if (!hasPermission) {
      throw new TRPCError({
        code: 'FORBIDDEN',
        message: `Missing permission: ${permissionCode}`,
      });
    }

    return opts.next({ ctx: { userId: ctx.userId } });
  });
```

**关键设计**：

- 工厂函数模式（`requirePermission(code)` 返回中间件），而非固定中间件
- 复用已有 `RbacModel.hasPermission`（内部已处理角色过期、isActive 等逻辑）
- 不修改 `authedProcedure`，保持现有路由不受影响

#### 4.1.2 导出 rbacProcedure

#### 4.1.3 新建 RBAC 管理路由

**文件**：`src/server/routers/lambda/rbac.ts`（新建）

参考 [apiKey.ts](src/server/routers/lambda/apiKey.ts) 的精确 router 模式：

```typescript
import { z } from 'zod';
import { RbacModel } from '@lobechat/database/models/rbac';
import { authedProcedure, rbacProcedure, router } from '@/libs/trpc/lambda';
import { serverDatabase } from '@/libs/trpc/lambda/middleware';

// 自定义 procedure：注入 RbacModel
const rbacManageProcedure = rbacProcedure('user:manage')
  .use(serverDatabase)
  .use(async (opts) => {
    const { ctx } = opts;
    return opts.next({
      ctx: { rbacModel: new RbacModel(ctx.serverDB, ctx.userId) },
    });
  });

// 只读查询：任何已认证用户可查自己的权限
const rbacQueryProcedure = authedProcedure.use(serverDatabase).use(async (opts) => {
  const { ctx } = opts;
  return opts.next({
    ctx: { rbacModel: new RbacModel(ctx.serverDB, ctx.userId) },
  });
});

export const rbacRouter = router({
  // 查自己的权限（任何用户可用）
  getMyPermissions: rbacQueryProcedure.query(async ({ ctx }) => {
    return ctx.rbacModel.getUserPermissionDetails();
  }),

  getMyRoles: rbacQueryProcedure.query(async ({ ctx }) => {
    return ctx.rbacModel.getUserRoles();
  }),

  // 以下需 user:manage 权限
  getRoles: rbacManageProcedure.query(async ({ ctx }) => {
    // 查全部角色（需扩展 RbacModel）
    return ctx.rbacModel.getAllRoles();
  }),

  getPermissions: rbacManageProcedure.query(async ({ ctx }) => {
    return ctx.rbacModel.getAllPermissions();
  }),

  getUserPermissions: rbacManageProcedure
    .input(z.object({ userId: z.string() }))
    .query(async ({ input, ctx }) => {
      return ctx.rbacModel.getUserPermissionDetails(input.userId);
    }),

  assignRole: rbacManageProcedure
    .input(z.object({ userId: z.string(), roleIds: z.array(z.string()) }))
    .mutation(async ({ input, ctx }) => {
      return ctx.rbacModel.updateUserRoles(input.userId, input.roleIds);
    }),
});
```

```typescript
import {
  boolean,
  index,
  integer,
  pgTable,
  primaryKey,
  text,
  uniqueIndex,
} from 'drizzle-orm/pg-core';
import { createdAt, timestamps } from './_helpers';
import { users } from './user';
import { createNanoId, idGenerator } from '../utils/idGenerator';

// ============ 用户组表 ============
export const userGroups = pgTable(
  'user_groups',
  {
    id: text('id')
      .$defaultFn(() => idGenerator('userGroups'))
      .primaryKey(),
    name: text('name').notNull(),
    description: text('description'),
    parentId: text('parent_id'), // 支持层级结构（自引用，后续加 FK）
    createdBy: text('created_by').references(() => users.id, { onDelete: 'set null' }),
    sort: integer('sort'),
    ...timestamps,
  },
  (t) => [
    index('user_groups_created_by_idx').on(t.createdBy),
    index('user_groups_parent_id_idx').on(t.parentId),
  ],
);

export type NewUserGroup = typeof userGroups.$inferInsert;
export type UserGroupItem = typeof userGroups.$inferSelect;

// ============ 用户组成员表 ============
export const userGroupMembers = pgTable(
  'user_group_members',
  {
    userId: text('user_id')
      .references(() => users.id, { onDelete: 'cascade' })
      .notNull(),
    groupId: text('group_id')
      .references(() => userGroups.id, { onDelete: 'cascade' })
      .notNull(),
    role: text('role').default('member').notNull(), // 'group_admin' | 'member'
    createdAt: createdAt(),
  },
  (t) => [
    primaryKey({ columns: [t.userId, t.groupId] }),
    index('user_group_members_user_id_idx').on(t.userId),
    index('user_group_members_group_id_idx').on(t.groupId),
  ],
);

export type NewUserGroupMember = typeof userGroupMembers.$inferInsert;

// ============ 用户上下级关系表 ============
export const userHierarchy = pgTable(
  'user_hierarchy',
  {
    managerId: text('manager_id')
      .references(() => users.id, { onDelete: 'cascade' })

    subordinateId: text('subordinate_id')
      .references(() => users.id, { onDelete: 'cascade' })
      .notNull(),
    createdAt: createdAt(),

    primaryKey({ columns: [t.managerId, t.subordinateId] }),
```

#### 4.2.5 新建 Model 层

**文件**：`packages/database/src/models/userGroup.ts`（新建）

精确复制 [SessionGroupModel](packages/database/src/models/sessionGroup.ts)（71 行）的模式：

```typescript
import { and, asc, desc, eq, inArray, sql } from 'drizzle-orm';
import { LobeChatDatabase } from '@/database/type';
import { idGenerator } from '../utils/idGenerator';
import {
  NewUserGroup,
  UserGroupItem,
  userGroupMembers,
  userGroups,
  userHierarchy,
} from '../schemas/userGroup';

export class UserGroupModel {
  private db: LobeChatDatabase;
  private userId: string;

  constructor(db: LobeChatDatabase, userId: string) {
    this.db = db;
    this.userId = userId;
  }

  // ---- 用户组 CRUD ----
  create = async (params: Pick<NewUserGroup, 'name' | 'description' | 'parentId'>) => {
    const [result] = await this.db
      .insert(userGroups)
      .values({ ...params, id: this.genId(), createdBy: this.userId })
      .returning();
    return result;
  };

  query = async () => {
    return this.db
      .select()
      .from(userGroups)
      .orderBy(asc(userGroups.sort), desc(userGroups.createdAt));
  };

  findById = async (id: string) => {
    return this.db.query.userGroups.findFirst({ where: eq(userGroups.id, id) });
  };

  update = async (id: string, value: Partial<UserGroupItem>) => {
    return this.db
      .update(userGroups)
      .set({ ...value, updatedAt: new Date() })
      .where(eq(userGroups.id, id));
  };

  delete = async (id: string) => {
    return this.db.delete(userGroups).where(eq(userGroups.id, id));
  };

  // ---- 成员管理 ----
  addMember = async (groupId: string, userId: string, role = 'member') => {
    return this.db.insert(userGroupMembers).values({ groupId, userId, role }).onConflictDoNothing();
  };

  removeMember = async (groupId: string, userId: string) => {
    return this.db
      .delete(userGroupMembers)
      .where(and(eq(userGroupMembers.groupId, groupId), eq(userGroupMembers.userId, userId)));
  };

  getGroupMembers = async (groupId: string) => {
    return this.db.select().from(userGroupMembers).where(eq(userGroupMembers.groupId, groupId));
  };

  getUserGroups = async (userId?: string) => {
    const targetUserId = userId || this.userId;
    return this.db
      .select({ groupId: userGroupMembers.groupId, role: userGroupMembers.role })
      .from(userGroupMembers)
      .where(eq(userGroupMembers.userId, targetUserId));
  };

  // ---- 上下级关系 ----
  addSubordinate = async (subordinateId: string) => {
    return this.db
      .insert(userHierarchy)
      .values({ managerId: this.userId, subordinateId })
      .onConflictDoNothing();
  };

  removeSubordinate = async (subordinateId: string) => {
    return this.db
      .delete(userHierarchy)
      .where(
        and(
          eq(userHierarchy.managerId, this.userId),
          eq(userHierarchy.subordinateId, subordinateId),
        ),
      );
  };

  getSubordinates = async (managerId?: string) => {
    const targetId = managerId || this.userId;
    return this.db
      .select({ subordinateId: userHierarchy.subordinateId })
      .from(userHierarchy)
      .where(eq(userHierarchy.managerId, targetId));
  };

  private genId = () => idGenerator('userGroups');
}
```

```typescript
    .where(inArray(topics.userId, userIds))
    .orderBy(desc(topics.updatedAt))
    .limit(size)
    .offset(page * size);
};

// ---- 话题锁定 ----
lockTopic = async (topicId: string, durationMinutes = 30) => {
  const expiresAt = new Date(Date.now() + durationMinutes * 60 * 1000);
  return this.db
    .insert(topicLocks)
    .values({ topicId, lockedBy: this.userId, expiresAt })
    .onConflictDoUpdate({
      target: topicLocks.topicId,
      set: { lockedBy: this.userId, lockedAt: new Date(), expiresAt },
      where: sql`${topicLocks.expiresAt} < NOW()`, // 只在已过期时覆盖
    });
};

unlockTopic = async (topicId: string) => {
  return this.db
    .delete(topicLocks)
    .where(and(eq(topicLocks.topicId, topicId), eq(topicLocks.lockedBy, this.userId)));
};

getTopicLock = async (topicId: string) => {
  const [lock] = await this.db
    .select()
    .from(topicLocks)
    .where(and(eq(topicLocks.topicId, topicId), sql`${topicLocks.expiresAt} > NOW()`))
    .limit(1);
  return lock ?? null;
};
```

**需导入**：`topicLocks` from `'../schemas/topicGroupShare'`

```typescript
import { z } from 'zod';
import { UserGroupModel } from '@lobechat/database/models/userGroup';
import { TopicModel } from '@lobechat/database/models/topic';
import { rbacProcedure, router } from '@/libs/trpc/lambda';
import { serverDatabase } from '@/libs/trpc/lambda/middleware';

const groupManageProcedure = rbacProcedure('user:manage')
  .use(serverDatabase)
  .use(async (opts) => {
    const { ctx } = opts;
    return opts.next({
      ctx: {
        userGroupModel: new UserGroupModel(ctx.serverDB, ctx.userId),
        topicModel: new TopicModel(ctx.serverDB, ctx.userId),
      },
    });
  });

export const userGroupRouter = router({
  // 用户组 CRUD
  createGroup: groupManageProcedure
    .input(
      z.object({
        name: z.string(),
        description: z.string().optional(),
        parentId: z.string().optional(),
      }),
    )
    .mutation(async ({ input, ctx }) => ctx.userGroupModel.create(input)),

  getGroups: groupManageProcedure.query(async ({ ctx }) => ctx.userGroupModel.query()),

  updateGroup: groupManageProcedure
    .input(
      z.object({ id: z.string(), name: z.string().optional(), description: z.string().optional() }),
    )
    .mutation(async ({ input: { id, ...value }, ctx }) => ctx.userGroupModel.update(id, value)),

  deleteGroup: groupManageProcedure
    .input(z.object({ id: z.string() }))
    .mutation(async ({ input, ctx }) => ctx.userGroupModel.delete(input.id)),

  // 成员管理
  addMember: groupManageProcedure
    .input(
      z.object({
        groupId: z.string(),
        userId: z.string(),
        role: z.enum(['group_admin', 'member']).optional(),
      }),
    )
    .mutation(async ({ input, ctx }) =>
      ctx.userGroupModel.addMember(input.groupId, input.userId, input.role),
    ),

  removeMember: groupManageProcedure
    .input(z.object({ groupId: z.string(), userId: z.string() }))
    .mutation(async ({ input, ctx }) =>
      ctx.userGroupModel.removeMember(input.groupId, input.userId),
    ),

  getGroupMembers: groupManageProcedure
    .input(z.object({ groupId: z.string() }))
    .query(async ({ input, ctx }) => ctx.userGroupModel.getGroupMembers(input.groupId)),

  // 层级关系
  addSubordinate: groupManageProcedure
    .input(z.object({ subordinateId: z.string() }))
    .mutation(async ({ input, ctx }) => ctx.userGroupModel.addSubordinate(input.subordinateId)),

  getSubordinates: groupManageProcedure.query(async ({ ctx }) =>
    ctx.userGroupModel.getSubordinates(),
  ),

  // 下属话题查看
  getSubordinateTopics: rbacProcedure('topic:view_subordinate')
    .use(serverDatabase)
    .use(async (opts) => {
      const { ctx } = opts;
      return opts.next({
        ctx: {
          userGroupModel: new UserGroupModel(ctx.serverDB, ctx.userId),
          topicModel: new TopicModel(ctx.serverDB, ctx.userId),
        },
      });
    })
    .input(z.object({ current: z.number().optional(), pageSize: z.number().optional() }))
    .query(async ({ input, ctx }) => {
      const subordinates = await ctx.userGroupModel.getSubordinates();
      const userIds = subordinates.map((s) => s.subordinateId);
      if (userIds.length === 0) return { items: [], total: 0 };
      const items = await ctx.topicModel.findBySubordinateUserIds(userIds, input);
      return { items, total: items.length };
    }),

  // 话题锁定
  lockTopic: rbacProcedure('topic:lock')
    .use(serverDatabase)
    .use(async (opts) => {
      const { ctx } = opts;
      return opts.next({ ctx: { topicModel: new TopicModel(ctx.serverDB, ctx.userId) } });
    })
    .input(z.object({ topicId: z.string(), durationMinutes: z.number().default(30) }))
    .mutation(async ({ input, ctx }) =>
      ctx.topicModel.lockTopic(input.topicId, input.durationMinutes),
    ),

  unlockTopic: rbacProcedure('topic:lock')
    .use(serverDatabase)
    .use(async (opts) => {
      const { ctx } = opts;
      return opts.next({ ctx: { topicModel: new TopicModel(ctx.serverDB, ctx.userId) } });
    })
    .input(z.object({ topicId: z.string() }))
    .mutation(async ({ input, ctx }) => ctx.topicModel.unlockTopic(input.topicId)),

  getTopicLock: rbacProcedure('topic:lock')
    .use(serverDatabase)
    .use(async (opts) => {
      const { ctx } = opts;
      return opts.next({ ctx: { topicModel: new TopicModel(ctx.serverDB, ctx.userId) } });
    })
    .input(z.object({ topicId: z.string() }))
    .query(async ({ input, ctx }) => ctx.topicModel.getTopicLock(input.topicId)),
});
```

```typescript
import { shallow } from 'zustand/shallow';
import { createWithEqualityFn } from 'zustand/traditional';
import { type StateCreator } from 'zustand/vanilla';
import { createDevtools } from '../middleware/createDevtools';

export interface UserGroupStore extends UserGroupState, UserGroupAction {}

interface UserGroupState {
  groups: any[];
  members: Record<string, any[]>;
  subordinates: any[];
  subordinateTopics: any[];
  loading: boolean;
}

interface UserGroupAction {
  fetchGroups: () => Promise<void>;
  createGroup: (params: { name: string; description?: string }) => Promise<void>;
  deleteGroup: (id: string) => Promise<void>;
  fetchMembers: (groupId: string) => Promise<void>;
  addMember: (groupId: string, userId: string) => Promise<void>;
  removeMember: (groupId: string, userId: string) => Promise<void>;
  fetchSubordinateTopics: () => Promise<void>;
}

const initialState: UserGroupState = {
  groups: [],
  members: {},
  subordinates: [],
  subordinateTopics: [],
  loading: false,
};

const createStore: StateCreator<UserGroupStore, [['zustand/devtools', never]]> = (set, get) => ({
  ...initialState,

  fetchGroups: async () => {
    set({ loading: true }, false, 'fetchGroups/start');
    const groups = await userGroupService.getGroups();
    set({ groups, loading: false }, false, 'fetchGroups/done');
  },

  createGroup: async (params) => {
    await userGroupService.createGroup(params);
    await get().fetchGroups();
  },

  deleteGroup: async (id) => {
    await userGroupService.deleteGroup(id);
    await get().fetchGroups();
  },

  fetchMembers: async (groupId) => {
    const members = await userGroupService.getGroupMembers(groupId);
    set((s) => ({ members: { ...s.members, [groupId]: members } }), false, 'fetchMembers');
  },

  addMember: async (groupId, userId) => {
    await userGroupService.addMember(groupId, userId);
    await get().fetchMembers(groupId);
  },

  removeMember: async (groupId, userId) => {
    await userGroupService.removeMember(groupId, userId);
    await get().fetchMembers(groupId);
  },

  fetchSubordinateTopics: async () => {
    const result = await userGroupService.getSubordinateTopics();
    set({ subordinateTopics: result.items }, false, 'fetchSubordinateTopics');
  },
});

const devtools = createDevtools('userGroup');
export const useUserGroupStore = createWithEqualityFn<UserGroupStore>()(
  devtools(createStore),
  shallow,
);
```
